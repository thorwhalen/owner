>>> b = int_to_bytes(an_integer)
>>> assert isinstance(b, bytes)
>>> back_to_int = bytes_to_int(b)
>>> assert back_to_int == an_integer

>>> int_to_bytes(7, n_bytes=3)
b'\x00\x00\x07'
>>> int_to_bytes(255, n_bytes=3)
b'\x00\x00\xff'
>>> int_to_bytes(256, n_bytes=3)
b'\x00\x01\x00'

b'\x00\x01\x07'
